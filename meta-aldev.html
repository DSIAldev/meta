# Generate-DataAldev.ps1 # Runbook Azure Automation (compte aa-aldev-metriques-stockage) - genere data-aldev.json # pour META-FLASH/META-NOW/META-TOUT (salles Aldev + bureaux flex b01-b12). # # PREREQUIS (a valider avant le 1er run reel) : # 1. Permission Graph Application "Calendars.Read" accordee a l'identite managee # + Application Access Policy scopee a un groupe de securite mail-enabled # contenant les salles Aldev ET b01@..b12@angers-developpement.com # (ne PAS utiliser Calendars.Read.All non scope sur tout le tenant) # 2. PAT GitHub (scope repo, idealement fine-grained limite a DSIAldev/meta) # stocke en Automation Variable chiffree "GitHubPAT_MetaRepo" # 3. Confirmer que b01-b12 sont bien des boites "Room" (pas "Equipment") - # sinon /places ne les remonte pas, voir bloc alternatif plus bas # # Reprend les acquis GAL-sync : [ordered]@{} partout, @() wrapping des collections, # Invoke-MgGraphRequest plutot que les cmdlets typees (contournement OutOfMemory/ # reflection deja rencontre sur Sites/Contacts), jamais de Connect-MgGraph avec # scopes custom dans un runbook (ici Identity uniquement, pas de conflit). $dryRun = $true # ne pousse rien sur GitHub tant que $true - inspecte $json d'abord Connect-MgGraph -Identity # --- 1. Liste des ressources (salles + bureaux), en dur ------------------- # Pas de requete dynamique (/places) : liste figee, a tenir a jour ici a la main # quand une salle/un bureau est ajoute ou renomme. # Note : pas de b04 (mailbox inexistante). Toutes les adresses uniformisees en bXX@ (primaire). $bureaux = @( [ordered]@{ displayName = 'Bureau 01'; emailAddress = 'b01@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 02'; emailAddress = 'b02@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 03'; emailAddress = 'b03@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 05'; emailAddress = 'b05@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 06'; emailAddress = 'b06@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 07'; emailAddress = 'b07@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 08'; emailAddress = 'b08@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 09'; emailAddress = 'b09@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 10'; emailAddress = 'b10@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 11'; emailAddress = 'b11@angers-developpement.com'; etage = $null } [ordered]@{ displayName = 'Bureau 12'; emailAddress = 'b12@angers-developpement.com'; etage = $null } ) $salles = @( # Le Cube 4 retire de la liste : salle fermee actuellement # Ordre : capacite decroissante (source liste ressources Outlook), puis alpha a egalite [ordered]@{ displayName = 'Scania'; emailAddress = 'scania@angers-developpement.com'; etage = $null } # 12 [ordered]@{ displayName = 'Bull'; emailAddress = 'bull@angers-developpement.com'; etage = $null } # 10 [ordered]@{ displayName = 'Le Cube 7'; emailAddress = 'lecube7@angers-developpement.com'; etage = $null } # 7 [ordered]@{ displayName = 'Le Cube 6'; emailAddress = 'lecube6@angers-developpement.com'; etage = $null } # 6 [ordered]@{ displayName = 'Bessonneau'; emailAddress = 'bessonneau@angers-developpement.com'; etage = $null } # 4 [ordered]@{ displayName = 'Bouvet'; emailAddress = 'bouvet@angers-developpement.com'; etage = $null } # 4 [ordered]@{ displayName = 'Ardoisières'; emailAddress = 'ardoisieres@angers-developpement.com'; etage = $null } # 3 [ordered]@{ displayName = 'Courant'; emailAddress = 'courant@angers-developpement.com'; etage = $null } # 3 ) $resources = @($salles) + @($bureaux) # --- 2. Fenetre de recherche ------------------------------------------------- $now = Get-Date $fin = $now.AddDays(60) # marge > 31j utilises par le front $headers = @{ Prefer = 'outlook.timezone="UTC"' } # evite tout calcul de decalage manuel # Invoke-MgGraphRequest deserialise parfois start.dateTime / end.dateTime en [datetime] # .NET (au lieu de laisser la string ISO brute) -> "$dt + 'Z'" tente alors un op_Addition # DateTime+TimeSpan et casse sur la conversion de 'Z'. On normalise donc dans les deux cas. function ConvertTo-IsoZ { param($value) if ($null -eq $value) { return $null } if ($value -is [datetime]) { return $value.ToString('yyyy-MM-ddTHH:mm:ss') + 'Z' } return [string]$value + 'Z' } # --- 3. Calendrier par ressource -------------------------------------------- $salles = foreach ($r in @($resources)) { $mail = $r.emailAddress if (-not $mail) { continue } $start = $now.ToString('yyyy-MM-ddTHH:mm:ss') $end = $fin.ToString('yyyy-MM-ddTHH:mm:ss') $uri = "https://graph.microsoft.com/v1.0/users/$mail/calendarView?startDateTime=$start&endDateTime=$end&`$top=250&`$select=subject,start,end,isAllDay,organizer" try { $events = Invoke-MgGraphRequest -Method GET -Uri $uri -Headers $headers # --- DEBUG : ce que Graph renvoie reellement, avant toute transformation --- Write-Output "DEBUG [$mail] uri=$uri" Write-Output "DEBUG [$mail] type brut=$($events.GetType().FullName)" Write-Output "DEBUG [$mail] cles racine=$(($events.Keys) -join ', ')" Write-Output "DEBUG [$mail] '@odata.count'=$($events.'@odata.count') '@odata.nextLink' presente=$([bool]$events.'@odata.nextLink')" Write-Output "DEBUG [$mail] value existe=$([bool]($events.ContainsKey('value'))) valueCount=$(@($events.value).Count)" if (@($events.value).Count -gt 0) { Write-Output "DEBUG [$mail] 1er evenement brut (non transforme) = $($events.value[0] | ConvertTo-Json -Depth 6 -Compress)" } else { Write-Output "DEBUG [$mail] reponse complete (value vide) = $($events | ConvertTo-Json -Depth 6 -Compress)" } # --- fin DEBUG --- $creneaux = @(foreach ($e in @($events.value)) { [ordered]@{ start = ConvertTo-IsoZ $e.start.dateTime # deja UTC grace au Prefer header end = ConvertTo-IsoZ $e.end.dateTime subject = $e.subject organizer = $e.organizer.emailAddress.address isallday = [bool]$e.isAllDay } }) Write-Output "DEBUG [$mail] creneaux transformes=$($creneaux.Count)" } catch { Write-Warning "Echec calendrier pour $mail : $($_.Exception.Message)" if ($_.ErrorDetails -and $_.ErrorDetails.Message) { Write-Warning "DEBUG [$mail] corps reponse erreur : $($_.ErrorDetails.Message)" } $creneaux = @() } [ordered]@{ nom = $r.displayName etage = $r.etage # renseigne a la main dans les listes en dur ci-dessus si besoin 'créneaux' = $creneaux } } $data = [ordered]@{ updated = (Get-Date).ToString('dd/MM/yyyy HH:mm') salles = @($salles) } $json = $data | ConvertTo-Json -Depth 6 -Compress if ($dryRun) { Write-Output $json return } # --- 4. Commit sur GitHub (repo DSIAldev/meta) ------------------------------ $pat = Get-AutomationVariable -Name 'GitHubPAT_MetaRepo' $owner = 'DSIAldev'; $repo = 'meta'; $path = 'data-aldev.json' $ghHeaders = @{ Authorization = "token $pat"; 'User-Agent' = 'aldev-automation' } $existing = $null try { $existing = Invoke-RestMethod -Method GET -Headers $ghHeaders ` -Uri "https://api.github.com/repos/$owner/$repo/contents/$path" } catch { } # 404 si le fichier n'existe pas encore -> creation simple $payload = [ordered]@{ message = "Update data-aldev.json - $(Get-Date -Format s)" content = [Convert]::ToBase64String([Text.Encoding]::UTF8.GetBytes($json)) } if ($existing) { $payload.sha = $existing.sha } Invoke-RestMethod -Method PUT -Headers $ghHeaders -ContentType 'application/json' ` -Uri "https://api.github.com/repos/$owner/$repo/contents/$path" ` -Body ($payload | ConvertTo-Json)
